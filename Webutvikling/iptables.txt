sudo apt update
#### install iptables + make persistent #
sudo apt install iptables iptables-persistent -y
####### enable ip forwarding ####
sudo nano etc/sysctl.conf
 net.ipv4.ip_forward=1
sudo sysctl -p
##### allow forward traffic ###

sudo iptables -A FORWARD -i eth1 -o eth0 -j ACCEPT
sudo iptables -A FORWARD -i eth0 -o eth1 -m state --state RELATED,ESTABLISHED -j ACCEPT

#### basic firewall ### 

sudo iptables -P INPUT DROP
sudo iptables -P FORWARD DROP
sudo iptables -P OUTPUT ACCEPT

#### allow essential traffic
sudo iptables -A INPUT -i lo -j ACCEPT
sudo iptables -A INPUT -m state --state RELATED,ESTABLISHED -j ACCEPT
sudo iptables -A INPUT -p tcp --dport 22 -j ACCEPT

####Check for time module
sudo modprobe xt_time

##### Timebased deny rule #######
sudo iptables -A FORWARD \
-m iprange --src-range 192.168.0.2-192.168.0.254 \
-m time --timestart 22:00 --timestop 06:00 \
--kerneltz \
-j DROP


##### default accept rule #### 
sudo iptables -A FORWARD \
-m iprange --src-range 192.168.1.100-192.168.1.200 \
-j ACCEPT



####### Save the rules
sudo netfilter-persistent save
####### Verify rules
sudo iptables -L -v -n
sudo iptables -L FORWARD -v -n

#### If you want to block internet only ####
sudo iptables -A FORWARD \
-i eth1 -o eth0 \
-m iprange --src-range 192.168.1.100-192.168.1.200 \
-m time --timestart 22:00 --timestop 06:00 \
--kerneltz \
-j DROP



sudo iptables -A FORWARD \
-m iprange --src-range 10.0.1.2-10.0.1.254 \
-m time --timestart 22:00 --timestop 06:00 \
--kerneltz \
-j DROP
